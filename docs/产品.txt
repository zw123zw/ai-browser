// 使用 sketch.js 或 jszip 自行解析
import JSZip from 'jszip'

async function parseSketchFile(file: File) {
  const zip = await JSZip.loadAsync(file)
  
  // Sketch 文件结构
  const meta = JSON.parse(await zip.file('meta.json').async('text'))
  const document = JSON.parse(await zip.file('document.json').async('text'))
  const pages = []
  
  // 读取所有页面
  for (const [name, data] of Object.entries(document.layers)) {
    const pageContent = JSON.parse(await zip.file(`pages/${name}.json`).async('text'))
    pages.push(pageContent)
  }
  
  // 提取图片资源
  const images = {}
  const imagesFolder = zip.folder('images')
  for (const [key, file] of Object.entries(imagesFolder.files)) {
    images[key] = await file.async('blob')
  }
  
  return { meta, pages, images }
}

import { fabric } from 'fabric'

class DesignRenderer {
  private canvas: fabric.Canvas
  private layers: fabric.Object[] = []
  
  constructor(canvasElement: HTMLCanvasElement) {
    this.canvas = new fabric.Canvas(canvasElement, {
      selection: false,
      renderOnAddRemove: false  // 批量渲染提升性能
    })
  }
  
  // 渲染整个设计稿
  async renderDocument(doc: SketchDocument) {
    this.canvas.clear()
    
    for (const page of doc.pages) {
      await this.renderLayer(page)
    }
    
    this.canvas.renderAll()
  }
  
  // 递归渲染图层
  async renderLayer(layer: SketchLayer) {
    switch (layer._class) {
      case 'rectangle':
        return this.renderRect(layer)
      case 'text':
        return this.renderText(layer)
      case 'group':
      case 'symbolMaster':
        return this.renderGroup(layer)
      case 'bitmap':
        return this.renderImage(layer)
      default:
        console.warn(`Unsupported layer type: ${layer._class}`)
    }
  }
  
  // 渲染矩形
  private renderRect(layer: SketchLayer): fabric.Rect {
    const rect = new fabric.Rect({
      left: layer.rect.x,
      top: layer.rect.y,
      width: layer.rect.width,
      height: layer.rect.height,
      fill: layer.style?.fills?.[0]?.color || '#ccc',
      selectable: false
    })
    this.canvas.add(rect)
    return rect
  }
  
  // 渲染文字
  private renderText(layer: SketchLayer): fabric.Text {
    const text = new fabric.Text(layer.attributedString?.string || '', {
      left: layer.rect.x,
      top: layer.rect.y,
      fontSize: layer.style?.fontSize || 14,
      fill: layer.style?.fontColor || '#000',
      fontFamily: layer.style?.fontFamily || 'Arial'
    })
    this.canvas.add(text)
    return text
  }
  
  // 渲染图片
  private async renderImage(layer: SketchLayer): Promise<fabric.Image> {
    const imgData = this.getImageData(layer.imageRef)
    const img = await fabric.util.loadImage(URL.createObjectURL(imgData))
    
    const fabricImg = new fabric.Image(img, {
      left: layer.rect.x,
      top: layer.rect.y,
      scaleX: layer.rect.width / img.width,
      scaleY: layer.rect.height / img.height
    })
    this.canvas.add(fabricImg)
    return fabricImg
  }
}

1. 缩放和平移
class CanvasInteraction {
  private canvas: fabric.Canvas
  private scale: number = 1
  private isDragging: boolean = false
  private lastX: number = 0
  private lastY: number = 0
  
  constructor(canvas: fabric.Canvas) {
    this.canvas = canvas
    this.initWheelZoom()
    this.initDragPan()
  }
  
  // 滚轮缩放
  private initWheelZoom() {
    this.canvas.on('mouse:wheel', (opt) => {
      const delta = opt.e.deltaY
      let zoom = this.canvas.getZoom()
      zoom *= 0.999 ** delta
      
      // 限制缩放范围
      zoom = Math.min(Math.max(0.1, zoom), 10)
      
      this.canvas.zoomToPoint({ x: opt.e.offsetX, y: opt.e.offsetY }, zoom)
      this.scale = zoom
      
      opt.e.preventDefault()
      opt.e.stopPropagation()
    })
  }
  
  // 拖拽平移
  private initDragPan() {
    this.canvas.on('mouse:down', (opt) => {
      if (opt.e.button === 1 || opt.e.altKey) {  // 中键或Alt+左键
        this.isDragging = true
        this.lastX = opt.e.clientX
        this.lastY = opt.e.clientY
        this.canvas.selection = false
      }
    })
    
    this.canvas.on('mouse:move', (opt) => {
      if (this.isDragging) {
        const vpt = this.canvas.viewportTransform
        vpt[4] += opt.e.clientX - this.lastX
        vpt[5] += opt.e.clientY - this.lastY
        this.canvas.requestRenderAll()
        this.lastX = opt.e.clientX
        this.lastY = opt.e.clientY
      }
    })
    
    this.canvas.on('mouse:up', () => {
      this.isDragging = false
      this.canvas.selection = true
    })
  }
}

class AnnotationTool {
  private canvas: fabric.Canvas
  private annotationGroup: fabric.Group | null = null
  
  constructor(canvas: fabric.Canvas) {
    this.canvas = canvas
    this.initClickAnnotation()
  }
  
  private initClickAnnotation() {
    this.canvas.on('mouse:down', (opt) => {
      if (opt.target) {
        this.showAnnotation(opt.target)
      } else {
        this.hideAnnotation()
      }
    })
  }
  
  private showAnnotation(target: fabric.Object) {
    this.hideAnnotation()
    
    const bounds = target.getBoundingRect()
    const lines: fabric.Line[] = []
    const texts: fabric.Text[] = []
    
    // 宽度标注线
    const widthLine = new fabric.Line(
      [bounds.left, bounds.top + bounds.height + 20, 
       bounds.left + bounds.width, bounds.top + bounds.height + 20],
      { stroke: '#ff4757', strokeWidth: 1 }
    )
    const widthText = new fabric.Text(`${Math.round(bounds.width)}`, {
      left: bounds.left + bounds.width / 2,
      top: bounds.top + bounds.height + 25,
      fontSize: 12,
      fill: '#ff4757',
      originX: 'center'
    })
    
    // 高度标注线
    const heightLine = new fabric.Line(
      [bounds.left + bounds.width + 20, bounds.top,
       bounds.left + bounds.width + 20, bounds.top + bounds.height],
      { stroke: '#2ed573', strokeWidth: 1 }
    )
    const heightText = new fabric.Text(`${Math.round(bounds.height)}`, {
      left: bounds.left + bounds.width + 25,
      top: bounds.top + bounds.height / 2,
      fontSize: 12,
      fill: '#2ed573',
      originY: 'center',
      angle: 90
    })
    
    this.annotationGroup = new fabric.Group(
      [widthLine, widthText, heightLine, heightText],
      { selectable: false, evented: false }
    )
    this.canvas.add(this.annotationGroup)
    this.canvas.renderAll()
  }
  
  private hideAnnotation() {
    if (this.annotationGroup) {
      this.canvas.remove(this.annotationGroup)
      this.annotationGroup = null
    }
  }
}

六、导出切图功能
async function exportSlice(layer: fabric.Object, format: 'png' | 'svg' = 'png') {
  if (format === 'png') {
    const dataURL = layer.toDataURL({
      format: 'png',
      multiplier: 2  // 2x 导出
    })
    
    // 触发下载
    const link = document.createElement('a')
    link.download = `${layer.name || 'slice'}.png`
    link.href = dataURL
    link.click()
  }
  
  if (format === 'svg') {
    const svg = layer.toSVG()
    const blob = new Blob([svg], { type: 'image/svg+xml' })
    const url = URL.createObjectURL(blob)
    
    const link = document.createElement('a')
    link.download = `${layer.name || 'slice'}.svg`
    link.href = url
    link.click()
    URL.revokeObjectURL(url)
  }
}
