/**
 * 文心一言 刷新AccessToken
 */
async function refreshWenxinToken() {
  const { clientId, clientSecret, tokenUrl } = currentModel;
  const url = `${tokenUrl}?grant_type=client_credentials&client_id=${clientId}&client_secret=${clientSecret}`;
  try {
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), REQUEST_TIMEOUT);
    const res = await fetch(url, { signal: controller.signal });
    clearTimeout(timer);

    const data = await res.json();
    currentModel.accessToken = data.access_token || "";
  } catch (err) {
    console.error("文心Token获取失败：", err);
    currentModel.accessToken = "";
  }
}

chrome.runtime.onMessage.addListener(async (msg, sender, sendResponse) => {
  if (msg.type !== "ai_parse") return true;

// ========== 模型调用函数 ==========
async function callQwen(prompt) {
  const { apiKey, chatUrl, modelName } = currentModel;
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), REQUEST_TIMEOUT);

  const body = JSON.stringify({
    model: modelName,
    input: { messages: [{ role: "user", content: prompt }] },
    parameters: { result_format: "text" }
  });

  const res = await fetch(chatUrl, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "Authorization": `Bearer ${apiKey}`
    },
    body,
    signal: controller.signal
  });
  clearTimeout(timer);

  const json = await res.json();
  return { code: 0, data: { result: json.output?.text || "" } };
}

async function callWenxin(prompt) {
  if (!currentModel.accessToken) return { code: -1, data: "文心Token失效" };
  const url = `${currentModel.chatUrl}?access_token=${currentModel.accessToken}`;
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), REQUEST_TIMEOUT);

  const body = JSON.stringify({ messages: [{ role: "user", content: prompt }] });
  const res = await fetch(url, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body,
    signal: controller.signal
  });
  clearTimeout(timer);

  const json = await res.json();
  return { code: 0, data: { result: json.result || "" } };
}

async function callXunfei(prompt) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), REQUEST_TIMEOUT);
  const body = JSON.stringify({ messages: [{ role: "user", content: prompt }] });

  const res = await fetch(currentModel.chatUrl, {

    headers: { "Content-Type": "application/json" },

    signal: controller.signal

  const json = await res.json();
  const content = json.choices?.[0]?.message?.content || "";
  return { code: 0, data: { result: content } };
}

async function callOllama(prompt) {
  const { chatUrl, modelName } = currentModel;
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), REQUEST_TIMEOUT);

  const body = JSON.stringify({

    method: "POST",
    headers: { "Content-Type": "application/json" },
    body,
    signal: controller.signal
  });
  clearTimeout(timer);

  const json = await res.json();
  return { code: 0, data: { result: json.response || "" } };
}

/**
 * 调用后台AI解析自然语言
 */
async function parseByAI(userText) {
  const systemPrompt = `
你是网页自动化控制助手，严格遵守规则：
1. 仅输出标准JSON字符串，**不要任何多余文字、解释、标点**
2. 支持动作列表：
open_site 打开网站，参数 url
search 全网搜索，参数 keyword
site_search 站内搜索，参数 keyword
open_item 点击第N个内容，参数 index(数字)
page_next 下一页
page_prev 上一页
video_play 播放视频
video_pause 暂停视频

示例：
用户：打开B站 → {"action":"open_site","url":"https://www.bilibili.com"}
用户：搜索美食教程 → {"action":"search","keyword":"美食教程"}
用户：打开第二个 → {"action":"open_item","index":2}

用户指令：${userText}
`;

  return new Promise((resolve) => {
    const timer = setTimeout(() => {
      resolve(null);
    }, AI_PARSE_TIMEOUT);

    chrome.runtime.sendMessage({ type: "ai_parse", prompt: systemPrompt }, (res) => {
      clearTimeout(timer);
      if (res.code !== 0) {
        resolve(null);
        return;
      }
      try {
        const cmdObj = JSON.parse(res.data.result);
        resolve(cmdObj);
      } catch (e) {
        resolve(null);
      }
    });
  });
}

/**
 * 执行自动化任务分发
 */
async function runAutoTask(cmd) {
  const { action, url, keyword, index } = cmd;
  switch (action) {
    case "open_site":
      chrome.tabs.create({ url });
      voiceBroadcast("已为你打开对应网页");
      break;
    case "search":
      const searchUrl = `https://www.baidu.com/s?wd=${encodeURIComponent(keyword)}`;
      chrome.tabs.create({ url: searchUrl }, (tab) => {
        injectScriptToTab(tab.id);
      });
      voiceBroadcast(`正在全网搜索：${keyword}`);
      break;
    case "site_search":
    case "open_item":
    case "page_next":
    case "page_prev":
    case "video_play":
    case "video_pause":
      await sendCmdToActiveTab(cmd);
      break;
    default:
      voiceBroadcast("暂不支持该操作");
  }
}

/**
 * 向当前激活标签页注入脚本并下发指令
 */
async function sendCmdToActiveTab(cmd) {
  const [activeTab] = await chrome.tabs.query({ active: true, currentWindow: true });
  await injectScriptToTab(activeTab.id);
  chrome.tabs.sendMessage(activeTab.id, cmd);
}

/**
 * 向指定标签页注入操控脚本
 */
function injectScriptToTab(tabId) {
  chrome.scripting.executeScript({
    target: { tabId },
    files: ["content.js"]
  });
}

  utter.rate = SPEAK_RATE;

    case "page_prev":
      handlePrevPage();
      break;
    case "video_play":
      handleVideoPlay(true);
      break;
    case "video_pause":
      handleVideoPlay(false);
      break;
  }
});

/**
 * 站内搜索
 */
function handleSiteSearch(keyword) {
  const searchInput = document.querySelector(
    "input[type='search'], input[name='wd'], input[placeholder*='搜索']," +
    ".search-input, .search-box input, #search-input"
  );
  if (!searchInput) return;

  searchInput.value = keyword;
  searchInput.dispatchEvent(new Event("input", { bubbles: true }));
  // 模拟回车提交
  const enterEvent = new KeyboardEvent("keydown", { key: "Enter", bubbles: true });
  searchInput.dispatchEvent(enterEvent);
}

/**
 * 点击第N个列表项（通用选择器，适配视频/资讯/电商/搜索结果）
 */
function handleClickItem(num) {
  const itemList = document.querySelectorAll(
    "h3 a, .item a, .card a, .result a, .title a, .list-item a," +
    ".video-item a, .feed-item a, .goods-item a"
  );
  const target = itemList[num - 1];
  if (!target) return;
  // 动态页面延时，等待DOM加载完成
  setTimeout(() => {
    target.click();
  }, 700);
}

/**
 * 下一页
 */
function handleNextPage() {
  const nextBtn = document.querySelector(
    ".next, .page-next, a[rel='next'], button.next," +
    "span:contains('下一页'), .pagination-next"
  );
  if (nextBtn) {
    nextBtn.click();
  } else {
    window.scrollTo({ top: document.body.scrollHeight, behavior: "smooth" });
  }
}

/**
 * 上一页
 */
function handlePrevPage() {
  const prevBtn = document.querySelector(
    ".prev, .page-prev, a[rel='prev'], span:contains('上一页'), .pagination-prev"
  );
  if (prevBtn) {
    prevBtn.click();
  } else {
    window.history.back();
  }
}

/**
 * 视频播放/暂停（全网HTML5视频通用）
 */
function handleVideoPlay(isPlay) {
  const video = document.querySelector("video");
  if (!video) return;
  if (isPlay) {
    video.play().catch(() => { });
  } else {
    video.pause();
  }
}

五、常见问题排查
语音无响应
检查麦克风权限、关闭麦克风独占软件、刷新扩展重试。
AI 解析失败
核对密钥、网络、CURRENT_MODEL 配置；Ollama 用户检查本地服务是否启动。
点击 / 搜索无效
小众网站 DOM 结构特殊，在 content.js 扩充选择器、加大延时。
请求超时
网络不佳，可在 background.js 调大 REQUEST_TIMEOUT 数值。
